const CLIP = DATA.clip;
const LEVELS = DATA.levels;
let audioUnlocked = false;

let state = {
  screen:'home', currentLevel:1, unlockedLevel:1, completedLevels:new Set(),
  q:0, score:0, history:[], answered:false, hasListenedOnce:false
};

function getLevel(id){ return LEVELS.find(l=>l.id===id); }
function clipSrc(){ return 'data:audio/mpeg;base64,' + CLIP; }

function render(){
  renderLevelMap();
  const screen = document.getElementById('screen');
  const hint = document.getElementById('hint');
  hint.textContent = '';
  if(state.screen === 'home') return renderHome(screen);
  if(state.screen === 'play') return renderPlay(screen, hint);
  if(state.screen === 'levelresult') return renderLevelResult(screen);
  if(state.screen === 'finished') return renderFinished(screen);
}

function renderHome(screen){
  const audioMsg = audioUnlocked
    ? `<p class="center" style="color:var(--green-ok);">🔊 Audio attivo.</p>`
    : `<p class="center" style="color:var(--text-dim);">Prima testa l'audio:</p>`;
  screen.innerHTML = `
    <div class="panel">
      <h2>Benvenuto/a!</h2>
      <p>Quattro livelli sullo stesso dialogo tra Lukas e Mia. Prima di rispondere a ogni livello dovrai ascoltare l'audio per intero almeno una volta. Per passare al livello successivo serve almeno l'80% di risposte corrette (8 su 10). Quando sbagli, ti mostro la frase del dialogo da cui si capiva la risposta giusta.</p>
      ${audioMsg}
      ${!audioUnlocked ? `<p class="center"><button class="btn-ghost" id="test-btn">🔈 Testa l'audio</button></p>` : ''}
      <p class="center"><button class="btn-primary" id="start-btn" ${!audioUnlocked?'disabled':''}>Inizia — Livello 1</button></p>
    </div>`;
  const testBtn = document.getElementById('test-btn');
  if(testBtn){
    testBtn.addEventListener('click', ()=>{
      const a = document.createElement('audio');
      a.src = clipSrc();
      a.play().then(()=>{ a.pause(); audioUnlocked = true; render(); }).catch(()=>{ audioUnlocked = true; render(); });
    });
  }
  document.getElementById('start-btn').addEventListener('click', ()=> startLevel(state.unlockedLevel));
}

function renderLevelMap(){
  const map = document.getElementById('levelmap');
  let html = '';
  LEVELS.forEach((lvl, idx)=>{
    let cls = 'node';
    if(state.completedLevels.has(lvl.id)) cls += ' done';
    else if(lvl.id === state.currentLevel && state.screen === 'play') cls += ' current';
    else if(lvl.id <= state.unlockedLevel) cls += ' unlocked';
    const label = state.completedLevels.has(lvl.id) ? '✓' : lvl.id;
    html += `<div class="node-wrap"><button class="${cls}" data-lvl="${lvl.id}" ${lvl.id<=state.unlockedLevel?'':'disabled'}>${label}</button>`;
    if(idx < LEVELS.length-1) html += `<div class="connector"></div>`;
    html += `</div>`;
  });
  map.innerHTML = html;
  map.querySelectorAll('.node.unlocked, .node.done').forEach(btn=>{
    btn.addEventListener('click', ()=>{
      const id = parseInt(btn.dataset.lvl,10);
      if(id <= state.unlockedLevel) startLevel(id);
    });
  });
}

function startLevel(id){
  state.screen='play'; state.currentLevel=id; state.q=0; state.score=0; state.history=[];
  state.answered=false; state.hasListenedOnce=false;
  render();
}

function renderPlay(screen, hint){
  const level = getLevel(state.currentLevel);
  const item = level.items[state.q];
  state.answered = false;

  let dots = '';
  for(let i=0;i<level.items.length;i++){
    let c='dot';
    if(state.history[i]==='ok') c+=' filled';
    if(state.history[i]==='ko') c+=' wrong';
    dots += `<div class="${c}"></div>`;
  }

  let bodyHtml = '';
  if(item.kind === 'mc'){
    bodyHtml = `
      <p class="center" style="font-weight:700;">${item.q}</p>
      <div class="options" id="options">
        ${item.opts.map((o,i)=>`<button class="option" data-idx="${i}" disabled>${o}</button>`).join('')}
      </div>`;
  } else {
    bodyHtml = `
      <p class="center" style="font-weight:700;">„${item.s}"</p>
      <div class="tf-row" id="options">
        <button class="option" data-idx="true" disabled>✅ Richtig</button>
        <button class="option" data-idx="false" disabled>❌ Falsch</button>
      </div>`;
  }

  screen.innerHTML = `
    <div class="partlabel">${level.short} — ${level.name}</div>
    <div class="levelsub">${level.info_it}</div>
    <div class="hud"><span>Domanda ${state.q+1}/${level.items.length}</span></div>
    <div class="progress-dots">${dots}</div>
    <div class="panel">
      <audio class="native-audio" controls id="audio-el" src="${clipSrc()}"></audio>
      <div class="listen-gate" id="listen-gate">${state.hasListenedOnce ? "✅ Hai già ascoltato — puoi rispondere." : "🔒 Ascolta l'audio per intero prima di poter rispondere."}</div>
      ${bodyHtml}
      <div class="feedback-box" id="feedback"></div>
    </div>`;

  const audioEl = document.getElementById('audio-el');
  const optionsDiv = document.getElementById('options');

  if(state.hasListenedOnce){
    optionsDiv.querySelectorAll('button').forEach(b=> b.disabled = false);
  }

  audioEl.addEventListener('ended', ()=>{
    if(!state.hasListenedOnce){
      state.hasListenedOnce = true;
      document.getElementById('listen-gate').textContent = '✅ Hai ascoltato — ora puoi rispondere.';
      document.getElementById('listen-gate').style.color = 'var(--green-ok)';
      optionsDiv.querySelectorAll('button').forEach(b=> b.disabled = false);
    }
  });

  optionsDiv.querySelectorAll('button').forEach(btn=>{
    btn.addEventListener('click', ()=> handleAnswer(btn.dataset.idx, item));
  });
}

function handleAnswer(rawVal, item){
  if(state.answered) return;
  state.answered = true;

  let ok;
  if(item.kind === 'mc'){
    ok = parseInt(rawVal,10) === item.correct;
  } else {
    ok = (rawVal === 'true') === item.correct;
  }
  state.history[state.q] = ok ? 'ok' : 'ko';
  if(ok) state.score++;

  document.getElementById('options').querySelectorAll('button').forEach(btn=>{
    btn.disabled = true;
    if(item.kind === 'mc'){
      const i = parseInt(btn.dataset.idx,10);
      if(i === item.correct) btn.classList.add('correct');
      else if(String(i) === rawVal) btn.classList.add('wrong');
    } else {
      const v = (btn.dataset.idx === 'true');
      if(v === item.correct) btn.classList.add('correct');
      else if(btn.dataset.idx === rawVal) btn.classList.add('wrong');
    }
  });

  const fb = document.getElementById('feedback');
  fb.classList.add('show');
  if(ok){
    fb.classList.add('ok'); fb.classList.remove('ko');
    fb.innerHTML = `<strong>Richtig!</strong>`;
  } else {
    fb.classList.add('ko'); fb.classList.remove('ok');
    fb.innerHTML = `<strong>Non corretto.</strong><div class="ref">Frase di riferimento: „${item.ref}"</div>`;
  }

  const level = getLevel(state.currentLevel);
  setTimeout(()=>{
    state.q++;
    if(state.q >= level.items.length) finishLevel();
    else renderPlay(document.getElementById('screen'), document.getElementById('hint'));
  }, ok ? 1600 : 3200);
}

function finishLevel(){
  const level = getLevel(state.currentLevel);
  const passed = state.score >= level.passScore;
  if(passed){
    state.completedLevels.add(level.id);
    if(level.id === state.unlockedLevel && level.id < LEVELS.length) state.unlockedLevel = level.id + 1;
  }
  state.screen = 'levelresult'; state.lastPassed = passed;
  render();
}

function renderLevelResult(screen){
  const level = getLevel(state.currentLevel);
  const isLast = level.id === LEVELS.length;
  const pct = Math.round((state.score/level.items.length)*100);
  if(state.lastPassed){
    screen.innerHTML = `
      <div class="panel">
        <h2>${isLast ? '🏆 Ultimo livello superato!' : 'Livello superato!'}</h2>
        <p class="center">${state.score} su ${level.items.length} corrette (${pct}%) — soglia dell'80% raggiunta.</p>
        <p class="center"><button class="btn-primary" id="cont-btn">${isLast ? 'Vai alla conclusione' : 'Livello successivo'}</button></p>
      </div>`;
    document.getElementById('cont-btn').addEventListener('click', ()=>{
      if(isLast){ state.screen='finished'; render(); }
      else startLevel(level.id+1);
    });
  } else {
    screen.innerHTML = `
      <div class="panel">
        <h2>Quasi! Serve l'80%</h2>
        <p class="center">${state.score} su ${level.items.length} corrette (${pct}%) — per superare questo livello servono almeno ${level.passScore} risposte corrette su ${level.items.length}.</p>
        <p class="center"><button class="btn-primary" id="retry-btn">Riprova il livello</button></p>
      </div>`;
    document.getElementById('retry-btn').addEventListener('click', ()=> startLevel(level.id));
  }
}

function renderFinished(screen){
  screen.innerHTML = `
    <div class="panel">
      <h2>🎉 Percorso completato!</h2>
      <p class="center">Hai superato tutti i livelli sul dialogo di Lukas e Mia, con almeno l'80% di risposte corrette in ciascuno.</p>
      <p class="center"><button class="btn-primary" id="restart-btn">Ricomincia da capo</button></p>
    </div>`;
  document.getElementById('restart-btn').addEventListener('click', ()=>{
    state = { screen:'home', currentLevel:1, unlockedLevel:1, completedLevels:new Set(), q:0, score:0, history:[], answered:false, hasListenedOnce:false };
    render();
  });
}
